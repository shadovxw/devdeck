manifest.yaml

id: sonarqube
category: quality
needs:
  - docker
connection_type:
  id: sonarqube
  fields:
    url:   { type: string, default: "http://localhost:9000" }
    token: { type: secret }
data_sources:
  projects: list Sonar projects (for repo settings dropdown)
actions:
  scan:
    inputs:  { repo_path: path, project_key: string, branch: string }
    outputs: { task_id: string, dashboard_url: string }
  quality_gate:
    inputs:  { task_id: string }
    outputs: { status: OK|ERROR, bugs: int, vulnerabilities: int }
    fail_when: status == ERROR
  issues:
    inputs:  { project_key: string }
    outputs: { findings: findings[] }      # goes to the Findings page
